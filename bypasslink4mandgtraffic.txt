        var step = function(now) {

            var dt = now - scrollState.lastTime;
            scrollState.lastTime = now;
            if (now < scrollState.pauseUntil) {
                scrollRAF = requestAnimationFrame(step);
                return;
            }
            var move = scrollState.step * (dt / 16);
            scrollState.offset += scrollState.dir * move;
            if (scrollState.offset >= scrollState.maxOffset) {

                scrollState.pauseUntil = now + 400;

                scrollState.dir = 1;
                scrollState.pauseUntil = now + 400;
            }
            try { window.scrollTo(0, Math.round(scrollState.originY + scrollState.offset)); } catch (e) {}
            scrollRAF = requestAnimationFrame(step);
        };
        scrollRAF = requestAnimationFrame(step);
    }

            var m = text.match(CD_PATTERNS[p]);
            if (m) {
                var sec = parseInt(m[1], 10);
                var phase = m[2] ? parseInt(m[2], 10) : 1;
                var total = m[3] ? parseInt(m[3], 10) : 2;
                if (!isNaN(sec) && sec >= 0 && sec <= 999) return { sec: sec, phase: phase, total: total, raw: m[0] };
            }
        }
        return null;
    }

    function readCountdownInDoc(doc) {
        if (!doc) return null;
        try { var bt = (doc.body && doc.body.textContent) || ''; var r1 = parseCountdownText(bt); if (r1) return r1; } catch (e) {}
        try {
            var sels = ['#timer','#countdown','#clock','.timer','.countdown','.clock','[id*="timer" i]','[id*="count" i]','[id*="clock" i]','[id*="wait" i]','[class*="timer" i]','[class*="count" i]','[class*="clock" i]','[class*="wait" i]'];
            for (var s = 0; s < sels.length; s++) {
                var els = doc.querySelectorAll(sels[s]);
                for (var e = 0; e < els.length; e++) {
                    var el = els[e];
                    if (el.closest && el.closest('#mp-root')) continue;
                    var t = (el.textContent || '').trim();
                    var r3 = parseCountdownText(t);
                    if (r3) return r3;
                    var num = t.match(/^(\d+)$/);
                    if (num) { var n = parseInt(num[1], 10); if (n >= 0 && n <= 999) return { sec: n, phase: 1, total: 2, raw: t }; }
                }
            }
        } catch (e) {}
        try {
            var walker = doc.createTreeWalker(doc.body || doc, NodeFilter.SHOW_TEXT, null, false);
            var node; var count = 0;
            while ((node = walker.nextNode()) && count < 5000) {
                count++;
                var text = (node.textContent || '').trim();
                if (!text || text.length < 3) continue;
                if (text.indexOf('ch\u1EDD') === -1 && text.indexOf('c\u00F2n') === -1 && text.indexOf('gi\u00E2y') === -1 && text.indexOf('s (') === -1 && text.indexOf('/') === -1) continue;
                var r4 = parseCountdownText(text);
                if (r4) return r4;
            }
        } catch (e) {}
        try {
            var iframes = doc.querySelectorAll('iframe');
            for (var f = 0; f < iframes.length; f++) {
                try { var cdoc = iframes[f].contentDocument || iframes[f].contentWindow.document; if (cdoc) { var r5 = readCountdownInDoc(cdoc); if (r5) return r5; } } catch (e) {}
            }
        } catch (e) {}
        try {
            var allEls = doc.querySelectorAll('*');
            for (var k = 0; k < allEls.length && k < 2000; k++) {
                var sr = allEls[k].shadowRoot;
                if (sr) { try { var r6 = readCountdownInDoc(sr); if (r6) return r6; } catch (e) {} }
            }
        } catch (e) {}
        return null;
    }

    function readServerCountdown() {
        var now = Date.now();
        var ttl = isLagOn() ? 500 : 250;
        if (now - _cache.cdT < ttl) return _cache.cd;
        var r = readCountdownInDoc(document);
        _cache.cd = r; _cache.cdT = now;
        return r;
    }

    // D\u00F9ng textContent (kh\u00F4ng g\u00E2y reflow) + cache
    function getFullTextAllDocs() {
        var now = Date.now();
        var ttl = isLagOn() ? 600 : 300;
        if (now - _cache.fullTextT < ttl) return _cache.fullText;
        var all = '';
        try { all += (document.body.textContent || '') + ' '; } catch (e) {}
        try {
            var iframes = document.querySelectorAll('iframe');
            for (var i = 0; i < iframes.length; i++) {
                try {
                    var cdoc = iframes[i].contentDocument || iframes[i].contentWindow.document;
                    if (cdoc && cdoc.body) all += (cdoc.body.textContent || '') + ' ';
                } catch (e) {}
            }
        } catch (e) {}
        _cache.fullText = all; _cache.fullTextT = now;
        return all;
    }

    function removeVietnameseAccents(s) {

    }

    function extractCode() {
        var all = getFullTextAllDocs();
        var patterns = [
            /m\u00E3\s*km[:\s]+([A-Za-z0-9_\-]{4,32})/i,
            /m\u00E3\s*code[:\s]+([A-Za-z0-9_\-]{4,32})/i,
            /m\u00E3\s*x\u00E1c\s*nh\u1EADn[:\s]+([A-Za-z0-9_\-]{4,32})/i,
            /m\u00E3\s*th\u01B0\u1EDFng[:\s]+([A-Za-z0-9_\-]{4,32})/i
        ];
        for (var p = 0; p < patterns.length; p++) {
            var m = all.match(patterns[p]);
            if (m) {
                var c = m[1].trim();
                if (/^https?:\/\//i.test(c)) continue;
                if (/\.(com|net|org|me|vn|io|info)/i.test(c)) continue;
                if (c.length < 4 || c.length > 32) continue;
                return c;
            }
        }
        // Buoc cuoi: quet cac nut/button co ma alphanumeric 4-10 ky tu (vd: w5WJWdQ4 hien tren nut xanh)
        // Chi chay khi khong con countdown (da het thoi gian cho)
        try {
            var scd = readServerCountdown();
            if (!scd || scd.sec === 0) {
                var cands = document.querySelectorAll('button, a, [role="button"], div, span');
                var skipWords = /^(submit|login|log in|send|ok|next|back|home|menu|close|open|start|stop|play|pause|search|buy|add|cart|like|share|follow|sub|join|get|go|new|old|yes|no|on|off|play|download|upload|view|edit|delete|remove|save|cancel|confirm|accept|decline|skip|done|finish|continue|loading|load|more|less|all|none|help|info|about|contact|settings|profile|account|logout|sign out|register|signup|sign up)$/i;
                for (var i2 = 0; i2 < cands.length; i2++) {
                    var b = cands[i2];
                    if (b.closest && b.closest('#mp-root')) continue;
                    if (b.offsetParent === null) continue;
                    var childCount = b.querySelectorAll('*').length;
                    if (childCount > 2) continue;
                    var txt = (b.textContent || '').replace(/\s+/g, ' ').trim();
                    if (!/^[A-Za-z0-9]{4,10}$/.test(txt)) continue;
                    if (skipWords.test(txt)) continue;
                    return txt;
                }
            }
        } catch (e) {}
        return null;
    }

    function findGoogleResult(domain) {
        var kw = domain.toLowerCase().replace(/^www\./, '');
        var links = Array.prototype.slice.call(document.querySelectorAll('a[href^="http"]'));
        for (var i = 0; i < links.length; i++) {
            var a = links[i];
            var h = a.href.toLowerCase();
            if (/google\.|gstatic\.|googleusercontent\.|youtube\.|wikipedia\.|facebook\.|accounts\./i.test(h)) continue;
            if (h.indexOf(kw) !== -1) return a;
        }
        var selectors = ['#search a[href^="http"]', '#rso a[href^="http"]', '.g a[href^="http"]', 'a[href^="http"]'];
        for (var s = 0; s < selectors.length; s++) {
            var els = document.querySelectorAll(selectors[s]);
            for (var e = 0; e < els.length; e++) {
                var el = els[e];
                var href = el.href.toLowerCase();
                if (/google\.|gstatic\.|googleusercontent\.|youtube\.|wikipedia\.|facebook\.|accounts\.|policies\.|support\.|maps\./i.test(href)) continue;
                if (href.indexOf('http') !== 0) continue;
                if (href.length < 15) continue;
                return el;
            }
        }
        return null;
    }

    function handleGtraffic() {
        // ===== Tu dong dien ma vao o (neu co ma va o nhap ma hien ra) =====
        var code = S.get('code');
        if (code) {
            var filled = S.get('filled', '0') === '1';
            var inp = findCodeInputOnLink4m();
            if (inp && !filled) {
                log('\uD83D\uDC36 GTRAFFIC - T\u1EF0 \u0110\u1ED8NG \u0110I\u1EC0N M\u00C3:', code);
                try {
                    inp.focus(); inp.value = code;
                    inp.dispatchEvent(new Event('input', { bubbles: true, cancelable: true }));
                    inp.dispatchEvent(new Event('change', { bubbles: true, cancelable: true }));
                    inp.dispatchEvent(new KeyboardEvent('keydown', { bubbles: true }));
                    inp.dispatchEvent(new KeyboardEvent('keyup', { bubbles: true }));
                    inp.dispatchEvent(new KeyboardEvent('keypress', { bubbles: true }));
                    inp.blur();
                    S.set('filled', '1');
                    log('\u2713 GTRAFFIC - \u0110\u00C3 \u0110I\u1EC0N M\u00C3 V\u00C0O \u00D4');
                    showCode(code);
                    showGuide('<b>\uD83D\uDC36 GTRAFFIC \u2014 \u0110\u00C3 \u0110I\u1EC0N M\u00C3 \u2192 ' + code + '</b>');
                    if (UI.status) UI.status.textContent = '\uD83D\uDC36 \u0110\u00E3 \u0111i\u1EC1n m\u00E3 v\u00E0o \u00F4';
                    setState('filling');
                } catch (e) { log('\u2717 L\u1ED7i \u0111i\u1EC1n m\u00E3:', e.message); }
            } else if (inp && filled) {
                if (UI.status) UI.status.textContent = '\uD83D\uDC36 \u0110\u00E3 \u0111i\u1EC1n m\u00E3 \u2014 gi\u1EA3i captcha & x\u00E1c nh\u1EADn';
                showCode(code);
            } else if (!inp && !filled) {
                if (UI.status) UI.status.textContent = '\uD83D\uDC36 Ch\u1EDD \u00F4 nh\u1EADp m\u00E3 xu\u1EA5t hi\u1EC7n...';
            }
        }
        return handleTarget();
    }

    function loop() {
        if (loopRunning) return;
        loopRunning = true;
        log('\u25B6 V\u00F2ng l\u1EB7p (poll=' + getPoll() + 'ms' + (isLagOn() ? ', gi\u1EA3m lag b\u1EADt' : '') + ')');
        var i = 0;
        var tick = function() {
            if (i >= CFG.maxLoops) { loopRunning = false; return; }
            i++;
            _cache.tick++;
            var p;
            if (isGoogle) p = handleGoogle();
            else if (isRealTargetPage()) p = handleTarget();
            else if (isGtraffic) p = handleGtraffic();
            else if (isLink4m) p = handleLink4m();
            else if (freeMode) p = handleTarget();
            else p = Promise.resolve();
            p.catch(function(e){ log('L\u1ED7i:', e.message); }).then(function() {
                if (getState() === 'done') { loopRunning = false; return; }
                setTimeout(tick, getPoll());
            });
        };
        tick();
    }

    function addResetButton() {
        if (!UI.actions) return;
        if (UI.actions.querySelector('.mp-reset-wrap')) return;
        var wrap = document.createElement('div');
        wrap.className = 'mp-btn-row mp-reset-wrap';
        var b = document.createElement('button');
        b.className = 'mp-btn gray';
        b.textContent = '\uD83D\uDC36 RESET TOOL';
        b.onclick = function() {
            if (!confirm('Reset to\u00E0n b\u1ED9?')) return;
            stopAutoScroll();
            S.reset();
            location.reload();
        };
        wrap.appendChild(b);
        UI.actions.appendChild(wrap);
    }

    function boot() {
        buildPanel();
        var st = getState();
        log('Kh\u1EDFi \u0111\u1ED9ng | Host:', HOST, '| State:', st, '| isRealTarget:', isRealTargetPage(), '| Gi\u1EA3m lag:', isLagOn() ? 'B\u1EACT' : 'T\u1EAET');
        refreshStatus();
        addResetButton();

        if (S.get('code')) showCode(S.get('code'));

        if (isRealTargetPage() && st === 'click-google') {
            setState('scan-click1');
        }
        if ((isGtraffic || freeMode) && (st === 'idle' || st === 'done')) {
            S.del('filled');
            setState('scan-click1');
        }

        if (freeMode || isGtraffic || (st && st !== 'idle' && st !== 'done')) {
            loop();
        }

        if (UI.bypassName) UI.bypassName.textContent = getBypassName();
        if (UI.note) {
            var bn = getBypassName();
            var md = getBtnMode() === 'g' ? 'N\u00DAT G' : 'N\u00DAT L\u1EA4Y M\u00C3';
            UI.note.textContent = '\uD83D\uDC36 BYPASS ' + bn + ' | ch\u1EBF: ' + md;
        }
        updateBtnModeUI();
    }

    if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', boot);
    else boot();
})();
