(function () {
    'use strict';

    var STORAGE_PREFIX = 'mp_gt_';

    function fallbackGet(key, def) {
        try {
            var raw = localStorage.getItem(STORAGE_PREFIX + key);
            return raw === null ? def : JSON.parse(raw);
        } catch (e) {
            return def;
        }
    }

    function fallbackSet(key, value) {
        try { localStorage.setItem(STORAGE_PREFIX + key, JSON.stringify(value)); } catch (e) {}
    }

    function fallbackDelete(key) {
        try { localStorage.removeItem(STORAGE_PREFIX + key); } catch (e) {}
    }

    // GM API wrappers: work in both managers, and still work if this file
    // is accidentally pasted directly into DevTools without GM APIs.
    if (typeof window.GM_addStyle !== 'function') {
        window.GM_addStyle = function (css) {
            var style = document.createElement('style');
            style.textContent = css || '';
            (document.head || document.documentElement).appendChild(style);
            return style;
        };
    }

    if (typeof window.GM_getValue !== 'function') {
        window.GM_getValue = function (key, def) { return fallbackGet(key, def); };
    }

    if (typeof window.GM_setValue !== 'function') {
        window.GM_setValue = function (key, value) { fallbackSet(key, value); };
    }

    if (typeof window.GM_deleteValue !== 'function') {
        window.GM_deleteValue = function (key) { fallbackDelete(key); };
    }

    window.S = window.S || {
        get: function (key, def) { return GM_getValue(STORAGE_PREFIX + key, def); },
        set: function (key, value) { GM_setValue(STORAGE_PREFIX + key, value); },
        del: function (key) { GM_deleteValue(STORAGE_PREFIX + key); },
        reset: function () {
            // Best-effort reset for keys used by this UI module.
            var keys = [
                'domain','keyword','gtrafficUrl','code','targetUrl','filled','confirmed',
                'lastClickTime','googleClickTries','codeFoundAt','lastServerSec',
                'serverTotalSec','navigatingBack','cdStartSec','cdStartAt','btnClickAt',
                'gotCodeAt','codeAttempts','backDone','lastNavUrl','lastNavAt','hardStop',
                'countdownRealZero','getLinkClicked','waitingGetLink','getLinkDeadline',
                'getLinkStartAt','pendingAutoReset','min','pos','lastCopyOk','lastCopyText'
            ];
            for (var i = 0; i < keys.length; i++) {
                try { GM_deleteValue(STORAGE_PREFIX + keys[i]); } catch (e) {}
            }
        }
    };

    window.STATE = window.STATE || {
        IDLE: 'idle',
        GOOGLE_SEARCH: 'google-search',
        GOOGLE_CLICK: 'google-click',
        SCAN_BTN: 'scan-btn',
        WAIT_COUNTDOWN: 'wait-countdown',
        GET_CODE: 'get-code',
        BACK_GTRAFFIC: 'back-gtraffic',
        FILL_CODE: 'fill-code',
        DONE: 'done'
    };

    window.getState = window.getState || function () {
        return S.get('state', STATE.IDLE);
    };

    window.setState = window.setState || function (state) {
        S.set('state', state);
        try { refreshStatus(); } catch (e) {}
    };

    window.resetFillFlags = window.resetFillFlags || function () {};

    window.log = window.log || function (msg) {
        try { console.log('[MP-GTRAFFIC]', msg); } catch (e) {}
    };

    window.IS_TOP = window.IS_TOP !== undefined ? window.IS_TOP : (function () {
        try { return window.top === window.self; } catch (e) { return true; }
    })();

    window.DOMAINS = window.DOMAINS || {
        _read: function () {
            var list = S.get('domains', []);
            return Array.isArray(list) ? list : [];
        },
        _write: function (list) { S.set('domains', list); },
        getList: function () { return this._read(); },
        save: function (value) {
            var d = String(value || '').trim();
            if (!d) return false;
            d = d.replace(/^https?:\/\//i, '').replace(/\/.*$/, '').trim();
            var list = this._read();
            if (list.some(function (x) { return x && x.d === d; })) return false;
            list.unshift({ d: d });
            if (list.length > 30) list = list.slice(0, 30);
            this._write(list);
            return true;
        },
        remove: function (value) {
            var d = String(value || '').trim();
            this._write(this._read().filter(function (x) { return !x || x.d !== d; }));
        }
    };
})();

function makeDomainRow(item) {
    var row = document.createElement('div');
    row.className = 'mp-domain-item';
    row.setAttribute('data-domain', item.d);
    var name = document.createElement('div');
    name.className = 'd-name';
    name.textContent = item.d;
    var useBtn = document.createElement('button');
    useBtn.className = 'd-use';
    useBtn.setAttribute('data-action', 'use');
    useBtn.textContent = 'DÙNG';
    var delBtn = document.createElement('button');
    delBtn.className = 'd-del';
    delBtn.setAttribute('data-action', 'del');
    delBtn.textContent = '✕';
    row.appendChild(name);
    row.appendChild(useBtn);
    row.appendChild(delBtn);
    return row;
}

// ============================================================
// 3. HAM XU LY CLICK TRONG DOMAIN LIST
// ============================================================
function onDomainListClick(e) {
    var target = e.target;
    if (!target || !target.closest) return;
    var row = target.closest('.mp-domain-item');
    if (!row) return;
    var d = row.getAttribute('data-domain');
    if (!d) return;
    var action = target.getAttribute && target.getAttribute('data-action');
    if (action === 'use') {
        e.stopPropagation();
        if (UI.domainInput) UI.domainInput.value = d;
        return;
    }
    if (action === 'del') {
        e.stopPropagation();
        DOMAINS.remove(d);
        renderDomainList();
        return;
    }
    if (UI.domainInput) UI.domainInput.value = d;
}

// 6. HAM ENABLE DRAG (KEO THA PANEL)
// ============================================================
function enableDrag() {
    var h = UI.head; if (!h) return;
    var dg = false, sx = 0, sy = 0, ox = 0, oy = 0;
    var dn = function(e) {
        if (e.target.closest('.mp-icon-btn')) return;
        if (UI.root.classList.contains('mp-min')) { toggleMin(); return; }
        dg = true;
        var p = e.touches ? e.touches[0] : e;
        sx = p.clientX; sy = p.clientY;
        var r = UI.root.getBoundingClientRect();
        ox = r.left; oy = r.top;
        UI.root.style.left = ox + 'px'; UI.root.style.top = oy + 'px'; UI.root.style.right = 'auto'; UI.root.style.transform = 'none';
        document.addEventListener('mousemove', mv, { passive: false });
        document.addEventListener('mouseup', up);
        document.addEventListener('touchmove', mv, { passive: false });
        document.addEventListener('touchend', up);
        e.preventDefault();
    };
    var mv = function(e) {
        if (!dg) return;
        var p = e.touches ? e.touches[0] : e;
        var nx = ox + (p.clientX - sx), ny = oy + (p.clientY - sy);
        nx = Math.max(0, Math.min(nx, window.innerWidth - UI.root.offsetWidth));
        ny = Math.max(0, Math.min(ny, window.innerHeight - UI.root.offsetHeight));
        UI.root.style.left = nx + 'px'; UI.root.style.top = ny + 'px';
        if (e.cancelable) e.preventDefault();
    };
    var up = function() {
        if (!dg) return;
        dg = false;
        var r = UI.root.getBoundingClientRect();
        S.set('pos', { x: r.left, y: r.top });
        document.removeEventListener('mousemove', mv);
        document.removeEventListener('mouseup', up);
        document.removeEventListener('touchmove', mv);
        document.removeEventListener('touchend', up);
    };
    h.addEventListener('mousedown', dn);
    h.addEventListener('touchstart', dn, { passive: false });
}

// 13. HAM COPY CLIPBOARD (DUNG CHO NUT COPY THU CONG)
// ============================================================
function copyToClipboard(text) {
    if (!text) return false;
    var tried = false;
    try {
        if (navigator.clipboard && navigator.clipboard.writeText) {
            navigator.clipboard.writeText(text).then(function() {
                log('COPY OK');
                S.set('lastCopyOk', '1');
                S.set('lastCopyText', text);
            }).catch(function(e) {
                if (copyViaExecCommand(text)) { S.set('lastCopyOk', '1'); S.set('lastCopyText', text); }
                else { S.set('lastCopyOk', '0'); S.set('lastCopyText', text); }
            });
            tried = true;
        }
    } catch (e) {}
    if (!tried) {
        if (copyViaExecCommand(text)) { S.set('lastCopyOk', '1'); S.set('lastCopyText', text); }
        else { S.set('lastCopyOk', '0'); S.set('lastCopyText', text); }
    }
    return true;
}

// ============================================================
// 14. HAM COPY FALLBACK (EXEC COMMAND)
// ============================================================
function copyViaExecCommand(text) {
    try {
        var ta = document.createElement('textarea');
        ta.value = text;
        ta.setAttribute('readonly', '');
        ta.style.cssText = 'position:fixed;top:0;left:0;width:2px;height:2px;padding:0;border:none;outline:none;box-shadow:none;background:transparent;opacity:0.01;z-index:-1;';
        document.body.appendChild(ta);
        ta.focus();
        ta.select();
        try { ta.setSelectionRange(0, text.length); } catch (e) {}
        var ok = false;
        try { ok = document.execCommand('copy'); } catch (e) {}
        document.body.removeChild(ta);
        if (ok) { log('COPY OK (exec)'); return true; }
        return false;
    } catch (e) { return false; }
}
