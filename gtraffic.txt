// ==UserScript==
// @name         zoanh extension GTraffic (Gộp)
// @namespace    taolambypass
// @version      3.0
// @description  Gộp toàn bộ Extension zoanh GTraffic thành Tampermonkey
// @author       iNoo
// @match        *://*/*
// @grant        GM_getValue
// @grant        GM_setValue
// @grant        GM_setClipboard
// @grant        unsafeWindow
// @run-at       document-start
// @connect      *
// ==/UserScript==

(function () {
  "use strict";

  // ==========================================
  // 1. QUẢN LÝ CẤU HÌNH (Thay cho chrome.storage)
  // ==========================================
  const Store = {
    get: (key, def) => GM_getValue(key, def),
    set: (key, val) => GM_setValue(key, val)
  };

  const state = {
    fixTab: Store.get("fixTab", false),
    fakeRef: Store.get("fakeRef", false),
    uiFixer: Store.get("uiFixer", false),
    autoClick: Store.get("autoClick", false)
  };

  // ==========================================
  // 2. SCRIPT FIX TAB (Chạy ngay từ document-start)
  // ==========================================
  function runFixTab() {
    if (!state.fixTab) return;
    const blockedEvents = ['visibilitychange', 'webkitvisibilitychange', 'blur', 'focusout', 'pagehide'];
    const originalAddEventListener = EventTarget.prototype.addEventListener;
    
    EventTarget.prototype.addEventListener = function(type, listener, options) {
      if (blockedEvents.includes(type)) return;
      return originalAddEventListener.apply(this, arguments);
    };
    
    const fakeState = { hidden: false, webkitHidden: false, visibilityState: 'visible', webkitVisibilityState: 'visible' };
    for (const [key, value] of Object.entries(fakeState)) {
      Object.defineProperty(document, key, { get: () => value, configurable: true });
    }
    document.hasFocus = () => true;
    window.onblur = null;
    unsafeWindow.requestAnimationFrame = function(callback) {
      return unsafeWindow.setTimeout(() => callback(performance.now()), 16);
    };
    unsafeWindow.cancelAnimationFrame = function(id) { unsafeWindow.clearTimeout(id); };
  }
  runFixTab();

  // ==========================================
  // 3. SCRIPT FAKE REFERRER
  // ==========================================
  function runFakeRef() {
    if (!state.fakeRef) return;
    // Cố gắng ghi đè document.referrer
    try {
      Object.defineProperty(document, 'referrer', {
        get: function() { return 'https://www.google.com/'; },
        configurable: true
      });
    } catch(e) {}
  }
  runFakeRef();

  // ==========================================
  // 4. UI PANEL & CSS (Thay cho popup.html)
  // ==========================================
  const panelCSS = `
    #zoanh-panel {
      position: fixed; bottom: 20px; right: 20px; width: 320px;
      background: #1c1c1f; color: #f2f2f4; border-radius: 12px;
      box-shadow: 0 10px 30px rgba(0,0,0,0.5); font-family: 'Segoe UI', Arial, sans-serif;
      z-index: 9999999; border: 1px solid #313136; overflow: hidden;
      transition: all 0.3s ease;
    }
    #zoanh-panel.minimized .zp-body { display: none; }
    .zp-header {
      display: flex; justify-content: space-between; align-items: center;
      padding: 12px 16px; background: #202024; border-bottom: 1px solid #313136;
      font-weight: bold; cursor: pointer; user-select: none;
    }
    .zp-header span { font-size: 14px; }
    .zp-minimize { background: none; border: none; color: #93939c; cursor: pointer; font-size: 18px; }
    .zp-body { padding: 16px; display: flex; flex-direction: column; gap: 10px; }
    .zp-toggle {
      display: flex; justify-content: space-between; align-items: center;
      background: #2a2a2f; padding: 10px 14px; border-radius: 8px; font-size: 13px;
    }
    .switch { position: relative; display: inline-block; width: 38px; height: 22px; }
    .switch input { opacity: 0; width: 0; height: 0; }
    .slider { position: absolute; cursor: pointer; top: 0; left: 0; right: 0; bottom: 0; background: #38383c; transition: .3s; border-radius: 22px; }
    .slider:before { position: absolute; content: ""; height: 16px; width: 16px; left: 3px; bottom: 3px; background: #d6d6da; transition: .3s; border-radius: 50%; }
    input:checked + .slider { background: #10b981; }
    input:checked + .slider:before { transform: translateX(16px); background: #fff; }
    .zp-btn {
      width: 100%; padding: 10px; background: #333338; color: #fff; border: none;
      border-radius: 8px; cursor: pointer; font-weight: bold; font-size: 13px; transition: 0.2s;
    }
    .zp-btn:hover { background: #444; }
    
    /* Link Redirect Modal */
    #zoanh-redirect-modal {
      position: fixed; inset: 0; background: rgba(0,0,0,0.7); z-index: 10000000;
      display: none; justify-content: center; align-items: center;
    }
    #zoanh-redirect-modal.show { display: flex; }
    .zr-content {
      background: #202024; width: 400px; border-radius: 12px; padding: 20px;
      max-height: 80vh; display: flex; flex-direction: column; color: #fff;
    }
    .zr-title { font-size: 16px; font-weight: bold; margin-bottom: 15px; display: flex; justify-content: space-between; }
    .zr-close { cursor: pointer; color: #f27171; font-size: 20px; }
    .zr-add-row { display: flex; gap: 8px; margin-bottom: 15px; }
    .zr-input { flex: 1; padding: 8px; background: #1c1c1f; border: 1px solid #313136; color: #fff; border-radius: 6px; outline: none; }
    .zr-list { overflow-y: auto; max-height: 300px; display: flex; flex-direction: column; gap: 8px; }
    .zr-item { display: flex; justify-content: space-between; align-items: center; background: #1c1c1f; padding: 10px; border-radius: 6px; font-size: 12px; }
    .zr-item span { word-break: break-all; flex: 1; }
    .zr-del { background: #f27171; color: #fff; border: none; padding: 4px 8px; border-radius: 4px; cursor: pointer; }
  `;

  function injectUI() {
    const style = document.createElement('style');
    style.textContent = panelCSS;
    document.head.appendChild(style);

    const panel = document.createElement('div');
    panel.id = 'zoanh-panel';
    panel.innerHTML = `
      <div class="zp-header" id="zp-header">
        <span>zoanh extension (TM)</span>
        <button class="zp-minimize" id="zp-minimize">−</button>
      </div>
      <div class="zp-body">
        <div class="zp-toggle">
          <span>Giữ tab hoạt động</span>
          <label class="switch"><input type="checkbox" id="toggle-fixTab"><span class="slider"></span></label>
        </div>
        <div class="zp-toggle">
          <span>Google Referrer</span>
          <label class="switch"><input type="checkbox" id="toggle-fakeRef"><span class="slider"></span></label>
        </div>
        <div class="zp-toggle">
          <span>Auto Click</span>
          <label class="switch"><input type="checkbox" id="toggle-autoClick"><span class="slider"></span></label>
        </div>
        <div class="zp-toggle">
          <span>UI Fixer</span>
          <label class="switch"><input type="checkbox" id="toggle-uiFixer"><span class="slider"></span></label>
        </div>
        <button class="zp-btn" id="open-redirect">Quản lý Link Redirect</button>
      </div>
    `;
    document.body.appendChild(panel);

    // Modal Quản lý Redirect
    const modal = document.createElement('div');
    modal.id = 'zoanh-redirect-modal';
    modal.innerHTML = `
      <div class="zr-content">
        <div class="zr-title">Find Link Redirect <span class="zr-close" id="zr-close">×</span></div>
        <div class="zr-add-row">
          <input class="zr-input" id="zr-id" placeholder="Mã ID" style="flex: 0.5;">
          <input class="zr-input" id="zr-url" placeholder="Link đích">
          <button class="zp-btn" id="zr-add" style="width: auto; padding: 0 15px;">+</button>
        </div>
        <div class="zr-list" id="zr-list"></div>
      </div>
    `;
    document.body.appendChild(modal);

    // Gắn sự kiện
    const header = document.getElementById('zp-header');
    const minimizeBtn = document.getElementById('zp-minimize');
    header.addEventListener('click', () => {
      panel.classList.toggle('minimized');
      minimizeBtn.textContent = panel.classList.contains('minimized') ? '+' : '−';
    });

    // Toggles
    const bindToggle = (id, key, callback) => {
      const el = document.getElementById(id);
      el.checked = state[key];
      el.addEventListener('change', (e) => {
        state[key] = e.target.checked;
        Store.set(key, e.target.checked);
        if (callback) callback(e.target.checked);
      });
    };
    bindToggle('toggle-fixTab', 'fixTab');
    bindToggle('toggle-fakeRef', 'fakeRef');
    bindToggle('toggle-autoClick', 'autoClick', (val) => { if (val) startAutoClick(); else stopAutoClick(); });
    bindToggle('toggle-uiFixer', 'uiFixer', (val) => { if (val) startUiFixer(); });

    // Modal Redirect Logic
    const modalEl = document.getElementById('zoanh-redirect-modal');
    document.getElementById('open-redirect').addEventListener('click', () => {
      modalEl.classList.add('show');
      renderRedirectList();
    });
    document.getElementById('zr-close').addEventListener('click', () => modalEl.classList.remove('show'));

    const getRedirectMap = () => JSON.parse(Store.get('lhRedirectMap', '{}'));
    const saveRedirectMap = (map) => Store.set('lhRedirectMap', JSON.stringify(map));

    function renderRedirectList() {
      const list = document.getElementById('zr-list');
      const map = getRedirectMap();
      list.innerHTML = '';
      if (Object.keys(map).length === 0) {
        list.innerHTML = '<div style="text-align:center;color:#666;font-size:12px;">Chưa có dữ liệu</div>';
        return;
      }
      for (const [id, url] of Object.entries(map)) {
        const item = document.createElement('div');
        item.className = 'zr-item';
        item.innerHTML = `<span><b>${id}</b>: ${url}</span><button class="zr-del" data-id="${id}">Xóa</button>`;
        list.appendChild(item);
      }
      list.querySelectorAll('.zr-del').forEach(btn => {
        btn.addEventListener('click', (e) => {
          const map = getRedirectMap();
          delete map[e.target.dataset.id];
          saveRedirectMap(map);
          renderRedirectList();
        });
      });
    }

    document.getElementById('zr-add').addEventListener('click', () => {
      const id = document.getElementById('zr-id').value.trim();
      let url = document.getElementById('zr-url').value.trim();
      if (!id || !url) return alert('Nhập thiếu thông tin!');
      if (!/^https?:\/\//i.test(url)) url = 'https://' + url;
      const map = getRedirectMap();
      map[id] = url;
      saveRedirectMap(map);
      document.getElementById('zr-id').value = '';
      document.getElementById('zr-url').value = '';
      renderRedirectList();
    });
  }

  // ==========================================
  // 5. AUTO CLICK & COPY MÃ (Thay cho script_autoclick.js)
  // ==========================================
  let autoClickInterval = null;
  let lastCopiedCode = "";

  function nativeClick(el) {
    ['mousedown', 'mouseup', 'click'].forEach(evt => {
      el.dispatchEvent(new MouseEvent(evt, { view: window, bubbles: true, cancelable: true }));
    });
  }

  function startAutoClick() {
    if (autoClickInterval) clearInterval(autoClickInterval);
    autoClickInterval = setInterval(() => {
      if (!state.autoClick) return;
      
      // Click nút G
      const btn = document.querySelector("#avt-btn");
      if (btn) nativeClick(btn);

      // Copy mã
      const codeContainer = document.querySelector('.trade-btn__content.copy-allowed, [class*="copy-allowed"]');
      if (codeContainer) {
        const text = codeContainer.textContent.replace(/\s+/g, '').trim();
        if (text && text !== lastCopiedCode && text.length > 3) {
          GM_setClipboard(text);
          lastCopiedCode = text;
          codeContainer.style.setProperty("background-color", "#28a745", "important");
          console.log("✅ Auto Copy:", text);
        }
      }
    }, 1500);
  }

  function stopAutoClick() {
    if (autoClickInterval) clearInterval(autoClickInterval);
    autoClickInterval = null;
    lastCopiedCode = "";
  }

  // ==========================================
  // 6. UI FIXER (Thay cho script_uifixer.js)
  // ==========================================
  let uiFixerInterval = null;

  function startUiFixer() {
    if (uiFixerInterval) clearInterval(uiFixerInterval);
    uiFixerInterval = setInterval(() => {
      if (!state.uiFixer) return;

      // Căn giữa nút AVT
      const btn = document.querySelector("#avt-btn");
      if (btn) {
        btn.style.setProperty("position", "fixed", "important");
        btn.style.setProperty("top", "50%", "important");
        btn.style.setProperty("left", "50%", "important");
        btn.style.setProperty("transform", "translate(-50%, -50%)", "important");
        btn.style.setProperty("z-index", "2147483647", "important");
        btn.style.setProperty("outline", "3px solid red", "important");
        btn.style.setProperty("outline-offset", "3px", "important");
      }

      // Xóa ngoặc vuông
      document.querySelectorAll('span, p, b, strong, h1, h2, h3, a, button').forEach(el => {
        if (el.children.length === 0 && el.innerText && (el.innerText.includes('[') || el.innerText.includes(']'))) {
          el.innerText = el.innerText.replace(/\[|\]/g, '');
        }
      });
    }, 500);
  }

  // ==========================================
  // 7. LINK REDIRECT (Thay cho script_lhredirect.js)
  // ==========================================
  function initLinkRedirect() {
    const checkAndRedirect = () => {
      const match = window.location.pathname.match(/^\/([a-zA-Z0-9_-]+)/);
      if (!match) return;
      const id = match[1];
      const map = JSON.parse(Store.get('lhRedirectMap', '{}'));
      
      if (map[id]) {
        setTimeout(() => {
          window.open(map[id], '_blank');
        }, 1000);
      }
    };
    setTimeout(checkAndRedirect, 100);
    let lastUrl = location.href;
    setInterval(() => {
      if (location.href !== lastUrl) {
        lastUrl = location.href;
        checkAndRedirect();
      }
    }, 500);
  }

  // ==========================================
  // KHỞI CHẠY
  // ==========================================
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', () => {
      injectUI();
      if (state.autoClick) startAutoClick();
      if (state.uiFixer) startUiFixer();
      if (location.hostname.includes('gtraffic.io')) initLinkRedirect();
    });
  } else {
    injectUI();
    if (state.autoClick) startAutoClick();
    if (state.uiFixer) startUiFixer();
    if (location.hostname.includes('gtraffic.io')) initLinkRedirect();
  }

})();
